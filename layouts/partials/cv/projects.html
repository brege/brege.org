{{/* Collapse blank-line runs so each application record is one block with summary lines, a stack line, and optional rail links. */}} {{- $blocks := split (replaceRE `\n{2,}` "\f" .body) "\f" -}}
{{- range $blocks -}} {{- $block := trim . "\r\n\t " -}} {{- if ne $block "" -}} {{- $lines := split $block "\n" -}} {{- if eq (len $lines) 1 -}}

{{ $block | markdownify }}

{{- continue -}} {{- end -}} {{- $stack := "" -}} {{- $links := slice -}} {{- $summaryLines := $lines -}} {{- if gt (len $lines) 1 -}} {{- $stack = trim (index $lines 1) "\r\n\t " -}} {{- $summaryLines = first 1 $lines -}} {{- end -}} {{- if gt (len $lines) 2 -}} {{- $links = after 2 $lines -}} {{- end -}} {{- $summary := trim (delimit $summaryLines " ") "\r\n\t " -}}
{{- $summary | markdownify -}}
{{- if ne $stack "" -}}
{{ $stack }}
{{- end -}}
{{- range $links -}} {{- $link := trim . "\r\n\t " -}} {{- if ne $link "" -}} {{ $link | markdownify }} {{- end -}} {{- end -}}
{{- end -}} {{- end -}}
